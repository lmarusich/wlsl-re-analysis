`summarise()` has grouped output by 'subject', 'condition'. You can override
using the `.groups` argument.
sub_ests_plot <- sub_mean_ests %>%
  group_by(condition, stim_distance) %>%
  summarize(group_mean_est = mean(sub_mean_est),
            group_n = n(),
            group_se = std.error(sub_mean_est),
            group_ci = qt(0.975, df = group_n - 1) * group_se) %>%
  ggplot(aes(x = stim_distance,
             y = group_mean_est,
             ymin = group_mean_est - group_ci,
             ymax = group_mean_est + group_ci,
             group = condition,
             color = condition)) +
  geom_line() +
  geom_point() +
  geom_errorbar() +
  labs(x = "Actual Distance", y = "Estimated Distance") +
  theme_classic() +
  geom_abline(aes(intercept = 0, slope = 1), linetype = 2) +
  scale_y_continuous(limits = c(0, 150)) +
  scale_colour_manual(values = friendly_pal("contrast_three"),
                      labels=c('Screen-locked', 'World-locked')) +
  theme(legend.title = element_blank())
